    24→/**
    25→ * Inpaint image using fast marching method with multi-pass approach
    26→ * for higher quality reconstruction
    27→ */
    28→export async function inpaintImage(
    29→  inputBuffer: Buffer,
    30→  maskBuffer: Buffer,
    31→  radius: number = 8
    32→): Promise<Buffer> {
    33→  const inputImage = sharp(inputBuffer);
    34→  const maskImage = sharp(maskBuffer);
    35→
    36→  const inputMeta = await inputImage.metadata();
    37→  const width = inputMeta.width!;
    38→  const height = inputMeta.height!;
    39→
    40→  // Get raw pixel data
    41→  const inputRaw = await inputImage
    42→    .ensureAlpha()
    43→    .raw()
    44→    .toBuffer();
    45→
    46→  const maskRaw = await maskImage
    47→    .resize(width, height, { fit: "fill" })
    48→    .ensureAlpha()
    49→    .raw()
    50→    .toBuffer();
    51→
    52→  // Create pixel arrays
    53→  const pixels: Pixel[] = [];
    54→  const isMasked: boolean[] = [];
    55→
    56→  for (let i = 0; i < width * height; i++) {
    57→    const idx = i * 4;
    58→    pixels.push({
    59→      r: inputRaw[idx],
    60→      g: inputRaw[idx + 1],
    61→      b: inputRaw[idx + 2],
    62→      a: inputRaw[idx + 3],
    63→    });
    64→    // Mask: white pixels (high value) = area to remove
    65→    const maskVal = (maskRaw[idx] + maskRaw[idx + 1] + maskRaw[idx + 2]) / 3;
    66→    isMasked.push(maskVal > 128);
    67→  }
    68→
    69→  // Multi-pass inpainting for better quality
    70→  // Pass 1: Initial reconstruction with larger radius for context
    71→  let outputPixels = await inpaintPass(pixels, isMasked, width, height, radius * 2);
    72→
    73→  // Pass 2: Refinement with smaller radius for detail
    74→  outputPixels = await inpaintPass(outputPixels, isMasked, width, height, radius);
    75→
    76→  // Pass 3: Smoothing on boundary for seamless blending
    77→  const smoothedPixels = applyBoundarySmoothing(outputPixels, isMasked, width, height, 3);
    78→
    79→  // Convert back to raw buffer
    80→  const outputRaw = Buffer.alloc(width * height * 4);
    81→  for (let i = 0; i < width * height; i++) {
    82→    const idx = i * 4;
    83→    outputRaw[idx] = smoothedPixels[i].r;
    84→    outputRaw[idx + 1] = smoothedPixels[i].g;
    85→    outputRaw[idx + 2] = smoothedPixels[i].b;
    86→    outputRaw[idx + 3] = smoothedPixels[i].a;
    87→  }
    88→
    89→  return sharp(outputRaw, { raw: { width, height, channels: 4 } })
    90→    .png({ quality: 100 })
    91→    .toBuffer();
    92→}

    94→/**
    95→ * Single inpainting pass using fast marching
    96→ */
    97→async function inpaintPass(
    98→  pixels: Pixel[],
    99→  isMasked: boolean[],
   100→  width: number,
   101→  height: number,
   102→  radius: number
   103→): Promise<Pixel[]> {
   104→  const known = new Uint8Array(width * height);
   105→  const distance = new Float32Array(width * height);
   106→  const processed = new Uint8Array(width * height);
   107→  const outputPixels: Pixel[] = [...pixels];
   108→
   109→  // Initialize known/unknown
   110→  for (let i = 0; i < width * height; i++) {
   111→    if (!isMasked[i]) {
   112→      known[i] = 1;
   113→      distance[i] = 0;
   114→    } else {
   115→      known[i] = 0;
   116→      distance[i] = Infinity;
   117→    }
   118→  }
   119→
   120→  // Find initial boundary
   121→  const boundary: number[] = [];
   122→  for (let y = 0; y < height; y++) {
   123→    for (let x = 0; x < width; x++) {
   124→      const idx = y * width + x;
   125→      if (!known[idx]) {
   126→        let hasKnownNeighbor = false;
   127→        for (const [dx, dy] of [[-1, 0], [1, 0], [0, -1], [0, 1]]) {
   128→          const nx = x + dx;
   129→          const ny = y + dy;
   130→          if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
   131→            if (known[ny * width + nx]) {
   132→              hasKnownNeighbor = true;
   133→              break;
   134→            }
   135→          }
   136→        }
   137→        if (hasKnownNeighbor) {
   138→          boundary.push(idx);
   139→          distance[idx] = 1;
   140→        }
   141→      }
   142→    }
   143→  }
   144→
   145→  // Sort by distance
   146→  const processQueue: number[] = [...boundary].sort((a, b) => distance[a] - distance[b]);
   147→
   148→  // Process using fast marching
   149→  const queueSet = new Set(boundary);
   150→
   151→  while (processQueue.length > 0) {
   152→    // Find the pixel with smallest distance
   153→    let minIdx = 0;
   154→    let minDist = Infinity;
   155→    for (let i = 0; i < processQueue.length; i++) {
   156→      if (distance[processQueue[i]] < minDist) {
   157→        minDist = distance[processQueue[i]];
   158→        minIdx = i;
   159→      }
   160→    }
   161→
   162→    const idx = processQueue.splice(minIdx, 1)[0];
   163→    queueSet.delete(idx);
   164→
   165→    // Skip if already processed (can happen if pixel was added multiple times)
   166→    if (processed[idx]) continue;
   167→
   168→    const x = idx % width;
   169→    const y = Math.floor(idx / width);
   170→
   171→    // Collect known pixels within radius using weighted interpolation
   172→    let totalR = 0, totalG = 0, totalB = 0, totalWeight = 0;
   173→
   174→    for (let dy = -radius; dy <= radius; dy++) {
   175→      for (let dx = -radius; dx <= radius; dx++) {
   176→        const nx = x + dx;
   177→        const ny = y + dy;
   178→        if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
   179→          const nIdx = ny * width + nx;
   180→          // Use known pixels OR already-processed (reconstructed) pixels
   181→          if (known[nIdx] || processed[nIdx]) {
   182→            const dist = Math.sqrt(dx * dx + dy * dy);
   183→            if (dist <= radius && dist > 0) {
   184→              // Inverse distance weighting
   185→              const weight = 1 / Math.pow(dist, 2);
   186→              // Direction preference: pixels closer to boundary get extra weight
   187→              const directionFactor = Math.max(0.1, 1 - distance[nIdx] / (radius * 2));
   188→              const finalWeight = weight * directionFactor;
   189→
   190→              totalR += outputPixels[nIdx].r * finalWeight;
   191→              totalG += outputPixels[nIdx].g * finalWeight;
   192→              totalB += outputPixels[nIdx].b * finalWeight;
   193→              totalWeight += finalWeight;
   194→            }
   195→          }
   196→        }
   197→      }
   198→    }
   199→
   200→    if (totalWeight > 0) {
   201→      outputPixels[idx] = {
   202→        r: Math.round(totalR / totalWeight),
   203→        g: Math.round(totalG / totalWeight),
   204→        b: Math.round(totalB / totalWeight),
   205→        a: 255,
   206→      };
   207→      processed[idx] = 1;
   208→      known[idx] = 1;
   209→
   210→      // Update distances of unknown neighbors and add to queue
   211→      for (const [ddx, ddy] of [[-1, 0], [1, 0], [0, -1], [0, 1]]) {
   212→        const nnx = x + ddx;
   213→        const nny = y + ddy;
   214→        if (nnx >= 0 && nnx < width && nny >= 0 && nny < height) {
   215→          const nnIdx = nny * width + nnx;
   216→          if (!known[nnIdx] && !processed[nnIdx] && !queueSet.has(nnIdx)) {
   217→            distance[nnIdx] = Math.min(distance[nnIdx], distance[idx] + 1);
   218→            processQueue.push(nnIdx);
   219→            queueSet.add(nnIdx);
   220→          }
   221→        }
   222→      }
   223→    }
   224→  }
   225→
   226→  return outputPixels;
   227→}

   229→/**
   230→ * Apply boundary smoothing for seamless blending
   231→ * Only smooths pixels at the boundary of mask regions
   232→ */
   233→function applyBoundarySmoothing(
   234→  pixels: Pixel[],
   235→  isMasked: boolean[],
   236→  width: number,
   237→  height: number,
   238→  radius: number
   239→): Pixel[] {
   240→  const output: Pixel[] = [...pixels];
   241→
   242→  for (let y = 0; y < height; y++) {
   243→    for (let x = 0; x < width; x++) {
   244→      const idx = y * width + x;
   245→      if (!isMasked[idx]) continue;
   246→
   247→      // Check if this is a boundary pixel
   248→      let isNearBoundary = false;
   249→      for (const [dx, dy] of [[-2, 0], [2, 0], [0, -2], [0, 2]]) {
   250→        const nx = x + dx;
   251→        const ny = y + dy;
   252→        if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
   253→          if (!isMasked[ny * width + nx]) {
   254→            isNearBoundary = true;
   255→            break;
   256→          }
   257→        }
   258→      }
   259→
   260→      if (!isNearBoundary) continue;
   261→
   262→      // Apply Gaussian-like smoothing
   263→      let totalR = 0, totalG = 0, totalB = 0, totalWeight = 0;
   264→
   265→      for (let dy = -radius; dy <= radius; dy++) {
   266→        for (let dx = -radius; dx <= radius; dx++) {
   267→          const nx = x + dx;
   268→          const ny = y + dy;
   269→          if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
   270→            const nIdx = ny * width + nx;
   271→            const dist = Math.sqrt(dx * dx + dy * dy);
   272→            if (dist <= radius) {
   273→              // Gaussian weight
   274→              const sigma = radius / 2;
   275→              const weight = Math.exp(-(dist * dist) / (2 * sigma * sigma));
   276→              totalR += pixels[nIdx].r * weight;
   277→              totalG += pixels[nIdx].g * weight;
   278→              totalB += pixels[nIdx].b * weight;
   279→              totalWeight += weight;
   280→            }
   281→          }
   282→        }
   283→      }
   284→
   285→      if (totalWeight > 0) {
   286→        output[idx] = {
   287→          r: Math.round(totalR / totalWeight),
   288→          g: Math.round(totalG / totalWeight),
   289→          b: Math.round(totalB / totalWeight),
   290→          a: 255,
   291→        };
   292→      }
   293→    }
   294→  }
   295→
   296→  return output;
   297→}

   303→interface DetectedRegion {

   311→/**
   312→ * Improved watermark detection
   313→ * Scans for semi-transparent regions, brightness anomalies in corners,
   314→ * and known watermark patterns (Gemini sparkle, text logos)
   315→ */
   316→export async function detectWatermark(
   317→  inputBuffer: Buffer
   318→): Promise<DetectedRegion[]> {
   319→  const image = sharp(inputBuffer);
   320→  const meta = await image.metadata();
   321→  const width = meta.width!;
   322→  const height = meta.height!;
   323→
   324→  const raw = await image.ensureAlpha().raw().toBuffer();
   325→
   326→  const regions: DetectedRegion[] = [];
   327→
   328→  // Strategy 1: Corner-based detection with brightness anomaly analysis
   329→  // Watermarks typically appear in corners as brighter or different-colored patches
   330→  // Use large search area (30% of min dimension) to ensure we catch the full watermark
   331→  const cornerSize = Math.min(width, height) * 0.30;
   332→  const corners = [
   333→    { x: width - cornerSize, y: height - cornerSize, name: 'bottom-right', confidence: 0.95 },
   334→    { x: 0, y: height - cornerSize, name: 'bottom-left', confidence: 0.7 },
   335→    { x: width - cornerSize, y: 0, name: 'top-right', confidence: 0.65 },
   336→    { x: 0, y: 0, name: 'top-left', confidence: 0.55 },
   337→  ];
   338→
   339→  for (const corner of corners) {
   340→    const regionStats = analyzeRegionDetailed(raw, width, height, corner.x, corner.y, cornerSize, cornerSize);
   341→
   342→    // Detect watermark if region has high brightness variation or transparency
   343→    if (regionStats.hasTransparency || regionStats.brightnessAnomaly > 0.12 || regionStats.colorDeviation > 0.18) {
   344→      // Use the full corner region as the mask (more reliable than refining)
   345→      // The inpainting algorithm will handle the non-watermark pixels gracefully
   346→      regions.push({
   347→        x: Math.round(corner.x),
   348→        y: Math.round(corner.y),
   349→        width: Math.round(cornerSize),
   350→        height: Math.round(cornerSize),
   351→        confidence: corner.confidence * (regionStats.hasTransparency ? 1.3 : 1),
   352→      });
   353→    }
   354→  }
   355→
   356→  // Strategy 2: Bottom edge scan (common for watermarks)
   357→  const bottomStripHeight = Math.min(height * 0.12, 80);
   358→  const bottomStripStats = analyzeRegionDetailed(raw, width, height, 0, height - bottomStripHeight, width, bottomStripHeight);
   359→  if (bottomStripStats.brightnessAnomaly > 0.1) {
   360→    // Use the full bottom strip as the mask region
   361→    regions.push({
   362→      x: 0,
   363→      y: Math.round(height - bottomStripHeight),
   364→      width: width,
   365→      height: Math.round(bottomStripHeight),
   366→      confidence: 0.75,
   367→    });
   368→  }
   369→
   370→  // Strategy 3: Detect Gemini-style sparkle (4-pointed star)
   371→  const sparkleRegion = detectGeminiSparkle(raw, width, height);
   372→  if (sparkleRegion) {
   373→    regions.push(sparkleRegion);
   374→  }
   375→
   376→  // Strategy 4: Detect text-like patterns (high contrast regions in corners)
   377→  const textRegions = detectTextPatterns(raw, width, height);
   378→  regions.push(...textRegions);
   379→
   380→  // Deduplicate and merge overlapping regions
   381→  let mergedRegions = mergeOverlappingRegions(regions);
   382→
   383→  // Fallback: Always ensure the bottom-right corner is covered
   384→  // This is the most common watermark position
   385→  const hasBottomRightCoverage = mergedRegions.some(r =>
   386→    r.x + r.width >= width * 0.85 && r.y + r.height >= height * 0.85
   387→  );
   388→  if (!hasBottomRightCoverage) {
   389→    const fallbackSize = Math.min(width, height) * 0.20;
   390→    mergedRegions.push({
   391→      x: Math.round(width - fallbackSize),
   392→      y: Math.round(height - fallbackSize),
   393→      width: Math.round(fallbackSize),
   394→      height: Math.round(fallbackSize),
   395→      confidence: 0.5,
   396→    });
   397→  }
   398→
   399→  // Sort by confidence
   400→  mergedRegions.sort((a, b) => b.confidence - a.confidence);
   401→
   402→  return mergedRegions;
   403→}

   405→/**
   406→ * Detailed region analysis
   407→ */
   408→function analyzeRegionDetailed(
   409→  raw: Buffer,
   410→  imgWidth: number,
   411→  imgHeight: number,
   412→  rx: number,
   413→  ry: number,
   414→  rw: number,
   415→  rh: number
   416→) {
   417→  let totalAlpha = 0;
   418→  let transparentCount = 0;
   419→  let colorSum = { r: 0, g: 0, b: 0 };
   420→  let colorVariance = { r: 0, g: 0, b: 0 };
   421→  let pixelCount = 0;
   422→  let brightnessSum = 0;
   423→  const colorValues: { r: number; g: number; b: number; brightness: number }[] = [];
   424→
   425→  const startX = Math.max(0, Math.floor(rx));
   426→  const startY = Math.max(0, Math.floor(ry));
   427→  const endX = Math.min(imgWidth, Math.ceil(rx + rw));
   428→  const endY = Math.min(imgHeight, Math.ceil(ry + rh));
   429→
   430→  for (let y = startY; y < endY; y++) {
   431→    for (let x = startX; x < endX; x++) {
   432→      const idx = (y * imgWidth + x) * 4;
   433→      const r = raw[idx];
   434→      const g = raw[idx + 1];
   435→      const b = raw[idx + 2];
   436→      const a = raw[idx + 3];
   437→
   438→      totalAlpha += a;
   439→      if (a < 255) transparentCount++;
   440→      colorSum.r += r;
   441→      colorSum.g += g;
   442→      colorSum.b += b;
   443→      const brightness = (r + g + b) / 3;
   444→      brightnessSum += brightness;
   445→      colorValues.push({ r, g, b, brightness });
   446→      pixelCount++;
   447→    }
   448→  }
   449→
   450→  if (pixelCount === 0) {
   451→    return { hasTransparency: false, colorDeviation: 0, uniformity: 0, brightnessAnomaly: 0 };
   452→  }
   453→
   454→  const avgAlpha = totalAlpha / pixelCount;
   455→  const avgColor = {
   456→    r: colorSum.r / pixelCount,
   457→    g: colorSum.g / pixelCount,
   458→    b: colorSum.b / pixelCount,
   459→  };
   460→  const avgBrightness = brightnessSum / pixelCount;
   461→
   462→  // Calculate color variance
   463→  let brightnessVarianceSum = 0;
   464→  for (const c of colorValues) {
   465→    colorVariance.r += (c.r - avgColor.r) ** 2;
   466→    colorVariance.g += (c.g - avgColor.g) ** 2;
   467→    colorVariance.b += (c.b - avgColor.b) ** 2;
   468→    brightnessVarianceSum += (c.brightness - avgBrightness) ** 2;
   469→  }
   470→
   471→  const variance = (colorVariance.r + colorVariance.g + colorVariance.b) / (pixelCount * 3);
   472→  const deviation = Math.sqrt(variance) / 255;
   473→  const brightnessVariance = Math.sqrt(brightnessVarianceSum / pixelCount) / 255;
   474→  const uniformity = 1 - deviation;
   475→
   476→  return {
   477→    hasTransparency: avgAlpha < 250 || transparentCount > pixelCount * 0.05,
   478→    colorDeviation: deviation,
   479→    uniformity,
   480→    brightnessAnomaly: brightnessVariance,
   481→    avgBrightness,
   482→  };
   483→}

   485→/**
   486→ * Refine the watermark region by finding the actual bounding box of anomalous pixels
   487→ * Uses robust background sampling from OUTSIDE the search region
   488→ */
   489→function refineWatermarkRegion(
   490→  raw: Buffer,
   491→  imgWidth: number,
   492→  imgHeight: number,
   493→  rx: number,
   494→  ry: number,
   495→  rw: number,
   496→  rh: number
   497→): { x: number; y: number; width: number; height: number } {
   498→  // Sample background from OUTSIDE the search region to avoid contamination
   499→  // Use a band around the search region
   500→  const bandSize = Math.max(10, Math.min(imgWidth, imgHeight) * 0.03);
   501→  const samplePoints: { x: number; y: number }[] = [];
   502→
   503→  // Sample from left of region
   504→  for (let y = Math.floor(ry); y < Math.min(ry + rh, imgHeight); y += Math.max(1, Math.floor(rh / 5))) {
   505→    for (let x = Math.max(0, Math.floor(rx - bandSize)); x < Math.floor(rx); x += 3) {
   506→      samplePoints.push({ x, y });
   507→    }
   508→  }
   509→  // Sample from above region
   510→  for (let x = Math.floor(rx); x < Math.min(rx + rw, imgWidth); x += Math.max(1, Math.floor(rw / 5))) {
   511→    for (let y = Math.max(0, Math.floor(ry - bandSize)); y < Math.floor(ry); y += 3) {
   512→      samplePoints.push({ x, y });
   513→    }
   514→  }
   515→  // Sample from right of region
   516→  for (let y = Math.floor(ry); y < Math.min(ry + rh, imgHeight); y += Math.max(1, Math.floor(rh / 5))) {
   517→    for (let x = Math.min(imgWidth - 1, Math.floor(rx + rw)); x < Math.min(imgWidth, Math.floor(rx + rw + bandSize)); x += 3) {
   518→      samplePoints.push({ x, y });
   519→    }
   520→  }
   521→
   522→  let bgR = 0, bgG = 0, bgB = 0, validSamples = 0;
   523→  for (const p of samplePoints) {
   524→    if (p.x >= 0 && p.x < imgWidth && p.y >= 0 && p.y < imgHeight) {
   525→      const idx = (p.y * imgWidth + p.x) * 4;
   526→      bgR += raw[idx];
   527→      bgG += raw[idx + 1];
   528→      bgB += raw[idx + 2];
   529→      validSamples++;
   530→    }
   531→  }
   532→
   533→  if (validSamples === 0) {
   534→    // Fallback: use center of image as background
   535→    const cx = Math.floor(imgWidth / 2);
   536→    const cy = Math.floor(imgHeight / 2);
   537→    const idx = (cy * imgWidth + cx) * 4;
   538→    bgR = raw[idx];
   539→    bgG = raw[idx + 1];
   540→    bgB = raw[idx + 2];
   541→  } else {
   542→    bgR /= validSamples;
   543→    bgG /= validSamples;
   544→    bgB /= validSamples;
   545→  }
   546→
   547→  // Find pixels that differ significantly from background
   548→  let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
   549→  let anomalousCount = 0;
   550→  const threshold = 20; // color difference threshold (lowered for sensitivity)
   551→
   552→  const startX = Math.max(0, Math.floor(rx));
   553→  const startY = Math.max(0, Math.floor(ry));
   554→  const endX = Math.min(imgWidth, Math.ceil(rx + rw));
   555→  const endY = Math.min(imgHeight, Math.ceil(ry + rh));
   556→
   557→  for (let y = startY; y < endY; y++) {
   558→    for (let x = startX; x < endX; x++) {
   559→      const idx = (y * imgWidth + x) * 4;
   560→      const r = raw[idx];
   561→      const g = raw[idx + 1];
   562→      const b = raw[idx + 2];
   563→      const a = raw[idx + 3];
   564→
   565→      const colorDiff = Math.sqrt(
   566→        (r - bgR) ** 2 + (g - bgG) ** 2 + (b - bgB) ** 2
   567→      );
   568→
   569→      // Detect anomalies: pixels that differ from background OR have transparency
   570→      if (colorDiff > threshold || a < 250) {
   571→        if (x < minX) minX = x;
   572→        if (x > maxX) maxX = x;
   573→        if (y < minY) minY = y;
   574→        if (y > maxY) maxY = y;
   575→        anomalousCount++;
   576→      }
   577→    }
   578→  }
   579→
   580→  if (anomalousCount === 0 || maxX < 0) {
   581→    return { x: Math.floor(rx), y: Math.floor(ry), width: Math.floor(rw), height: Math.floor(rh) };
   582→  }
   583→
   584→  // Add generous padding to ensure full watermark coverage
   585→  const padding = Math.max(8, Math.min(imgWidth, imgHeight) * 0.02);
   586→  return {
   587→    x: Math.max(0, Math.floor(minX - padding)),
   588→    y: Math.max(0, Math.floor(minY - padding)),
   589→    width: Math.min(imgWidth, Math.ceil(maxX - minX + padding * 2)),
   590→    height: Math.min(imgHeight, Math.ceil(maxY - minY + padding * 2)),
   591→  };
   592→}

   594→/**
   595→ * Detect Gemini-style sparkle watermark (4-pointed star)
   596→ */
   597→function detectGeminiSparkle(
   598→  raw: Buffer,
   599→  width: number,
   600→  height: number
   601→): DetectedRegion | null {
   602→  // Search bottom-right area for sparkle pattern
   603→  // Use large search area (25% of min dimension) to ensure full watermark coverage
   604→  const sparkleSearchSize = Math.min(width, height) * 0.25;
   605→  const startX = Math.floor(width - sparkleSearchSize);
   606→  const startY = Math.floor(height - sparkleSearchSize);
   607→  const searchW = Math.ceil(sparkleSearchSize);
   608→  const searchH = Math.ceil(sparkleSearchSize);
   609→
   610→  let sparklePixels = 0;
   611→  let totalPixels = 0;
   612→
   613→  for (let y = startY; y < Math.min(startY + searchH, height); y++) {
   614→    for (let x = startX; x < Math.min(startX + searchW, width); x++) {
   615→      const idx = (y * width + x) * 4;
   616→      const r = raw[idx];
   617→      const g = raw[idx + 1];
   618→      const b = raw[idx + 2];
   619→      const a = raw[idx + 3];
   620→
   621→      const brightness = (r + g + b) / 3;
   622→
   623→      // Sparkle pattern: semi-transparent, lighter than surrounding area
   624→      if ((a < 255 || brightness > 200) && brightness > 180) {
   625→        sparklePixels++;
   626→      }
   627→      totalPixels++;
   628→    }
   629→  }
   630→
   631→  if (totalPixels === 0) return null;
   632→
   633→  // If sparkle-like pixels found with sufficient density
   634→  if (sparklePixels > totalPixels * 0.03 && sparklePixels > 5) {
   635→    // Use the full search region as the mask (more reliable)
   636→    return {
   637→      x: startX,
   638→      y: startY,
   639→      width: searchW,
   640→      height: searchH,
   641→      confidence: 0.9,
   642→    };
   643→  }
   644→
   645→  return null;
   646→}

   648→/**
   649→ * Detect text-like patterns in image corners
   650→ * Looks for high-contrast regions that could be text watermarks
   651→ */
   652→function detectTextPatterns(
   653→  raw: Buffer,
   654→  width: number,
   655→  height: number
   656→): DetectedRegion[] {
   657→  const regions: DetectedRegion[] = [];
   658→  const cornerSize = Math.min(width, height) * 0.15;
   659→
   660→  const corners = [
   661→    { x: width - cornerSize, y: height - cornerSize, name: 'br' },
   662→    { x: 0, y: height - cornerSize, name: 'bl' },
   663→    { x: width - cornerSize, y: 0, name: 'tr' },
   664→  ];
   665→
   666→  for (const corner of corners) {
   667→    // Compute local contrast in this region
   668→    let contrastSum = 0;
   669→    let pixelCount = 0;
   670→    const blockX = Math.floor(corner.x);
   671→    const blockY = Math.floor(corner.y);
   672→    const blockW = Math.min(cornerSize, width - blockX);
   673→    const blockH = Math.min(cornerSize, height - blockY);
   674→
   675→    for (let y = blockY; y < blockY + blockH - 1; y++) {
   676→      for (let x = blockX; x < blockX + blockW - 1; x++) {
   677→        const idx = (y * width + x) * 4;
   678→        const idxRight = (y * width + x + 1) * 4;
   679→        const idxDown = ((y + 1) * width + x) * 4;
   680→
   681→        const b1 = (raw[idx] + raw[idx + 1] + raw[idx + 2]) / 3;
   682→        const b2 = (raw[idxRight] + raw[idxRight + 1] + raw[idxRight + 2]) / 3;
   683→        const b3 = (raw[idxDown] + raw[idxDown + 1] + raw[idxDown + 2]) / 3;
   684→
   685→        contrastSum += Math.abs(b1 - b2) + Math.abs(b1 - b3);
   686→        pixelCount++;
   687→      }
   688→    }
   689→
   690→    if (pixelCount === 0) continue;
   691→    const avgContrast = contrastSum / pixelCount;
   692→
   693→    // High contrast suggests text presence
   694→    if (avgContrast > 20) {
   695→      const refined = refineWatermarkRegion(raw, width, height, blockX, blockY, blockW, blockH);
   696→      regions.push({
   697→        x: refined.x,
   698→        y: refined.y,
   699→        width: refined.width,
   700→        height: refined.height,
   701→        confidence: 0.7,
   702→      });
   703→    }
   704→  }
   705→
   706→  return regions;
   707→}

   709→/**
   710→ * Merge overlapping regions to avoid duplicate processing
   711→ */
   712→function mergeOverlappingRegions(regions: DetectedRegion[]): DetectedRegion[] {
   713→  if (regions.length === 0) return regions;
   714→
   715→  const merged: DetectedRegion[] = [];
   716→  const used = new Array(regions.length).fill(false);
   717→
   718→  for (let i = 0; i < regions.length; i++) {
   719→    if (used[i]) continue;
   720→    let current = { ...regions[i] };
   721→    used[i] = true;
   722→
   723→    for (let j = i + 1; j < regions.length; j++) {
   724→      if (used[j]) continue;
   725→      if (regionsOverlap(current, regions[j])) {
   726→        current = mergeRegions(current, regions[j]);
   727→        used[j] = true;
   728→      }
   729→    }
   730→
   731→    merged.push(current);
   732→  }
   733→
   734→  return merged;
   735→}
   736→
   737→function regionsOverlap(a: DetectedRegion, b: DetectedRegion): boolean {
   738→  return !(
   739→    a.x + a.width < b.x ||
   740→    b.x + b.width < a.x ||
   741→    a.y + a.height < b.y ||
   742→    b.y + b.height < a.y
   743→  );
   744→}
   745→
   746→function mergeRegions(a: DetectedRegion, b: DetectedRegion): DetectedRegion {
   747→  const x = Math.min(a.x, b.x);
   748→  const y = Math.min(a.y, b.y);
   749→  const right = Math.max(a.x + a.width, b.x + b.width);
   750→  const bottom = Math.max(a.y + a.height, b.y + b.height);
   751→  return {
   752→    x,
   753→    y,
   754→    width: right - x,
   755→    height: bottom - y,
   756→    confidence: Math.max(a.confidence, b.confidence),
   757→  };
   758→}

   760→/**
   761→ * Generate a mask from detected watermark regions
   762→ * Adds generous padding for full coverage
   763→ */
   764→export async function generateDetectionMask(
   765→  inputBuffer: Buffer,
   766→  regions: DetectedRegion[]
   767→): Promise<Buffer> {
   768→  const image = sharp(inputBuffer);
   769→  const meta = await image.metadata();
   770→  const width = meta.width!;
   771→  const height = meta.height!;
   772→
   773→  // Create a white mask on black background
   774→  const maskRaw = Buffer.alloc(width * height * 4);
   775→
   776→  // Fill with black (no mask)
   777→  for (let i = 0; i < width * height; i++) {
   778→    const idx = i * 4;
   779→    maskRaw[idx] = 0;
   780→    maskRaw[idx + 1] = 0;
   781→    maskRaw[idx + 2] = 0;
   782→    maskRaw[idx + 3] = 255;
   783→  }
   784→
   785→  // Mark detected regions as white (mask)
   786→  for (const region of regions) {
   787→    for (let y = region.y; y < Math.min(region.y + region.height, height); y++) {
   788→      for (let x = region.x; x < Math.min(region.x + region.width, width); x++) {
   789→        const idx = (y * width + x) * 4;
   790→        maskRaw[idx] = 255;
   791→        maskRaw[idx + 1] = 255;
   792→        maskRaw[idx + 2] = 255;
   793→      }
   794→    }
   795→  }
   796→
   797→  // Add generous padding around mask regions for better coverage
   798→  const padding = Math.max(8, Math.min(width, height) * 0.015);
   799→  const paddedMask = addMaskPadding(maskRaw, width, height, Math.floor(padding));
   800→
   801→  return sharp(paddedMask, { raw: { width, height, channels: 4 } })
   802→    .png()
   803→    .toBuffer();
   804→}

   806→/**
   807→ * Add padding around mask regions to ensure full watermark coverage
   808→ */
   809→function addMaskPadding(
   810→  mask: Buffer,
   811→  width: number,
   812→  height: number,
   813→  padding: number
   814→): Buffer {
   815→  const result = Buffer.from(mask);
   816→
   817→  // For each white pixel, expand by padding
   818→  for (let y = 0; y < height; y++) {
   819→    for (let x = 0; x < width; x++) {
   820→      const idx = (y * width + x) * 4;
   821→      if (mask[idx] > 128) {
   822→        // Expand
   823→        for (let dy = -padding; dy <= padding; dy++) {
   824→          for (let dx = -padding; dx <= padding; dx++) {
   825→            const nx = x + dx;
   826→            const ny = y + dy;
   827→            if (nx >= 0 && nx < width && ny >= 0 && ny < height) {
   828→              const nIdx = (ny * width + nx) * 4;
   829→              result[nIdx] = 255;
   830→              result[nIdx + 1] = 255;
   831→              result[nIdx + 2] = 255;
   832→            }
   833→          }
   834→        }
   835→      }
   836→    }
   837→  }
   838→
   839→  return result;
   840→}

   987→  ctx.fillStyle = options.color;
   988→

   990→    // Repeat watermark across entire image

   997→        ctx.save();
   998→        ctx.translate(x + textWidth / 2, y + textHeight / 2);
   999→        ctx.rotate(rotationRad);
  1000→        ctx.fillText(options.text, -textWidth / 2, -textHeight / 2);
  1001→        ctx.restore();
  1002→      }
  1003→    }
  1004→  } else {
  1005→    // Single watermark at specified position
  1006→    const margin = Math.max(10, Math.min(canvasWidth, canvasHeight) * 0.03);

  1011→      ctx.rotate(options.rotation * Math.PI / 180);
  1012→      ctx.fillText(options.text, -textWidth / 2, -textHeight / 2);
  1013→    } else {

  1081→export async function optimizeImage(
  1082→  inputBuffer: Buffer,

  1087→  const originalSize = inputBuffer.length;

  1094→      fit: "inside",
  1095→      withoutEnlargement: true,

  1100→  switch (options.format) {
  1101→    case "jpeg":
  1102→      pipeline = pipeline.jpeg({
  1103→        quality: options.quality,
  1104→        mozjpeg: true,
  1105→      });

  1107→    case "png":

  1110→        compressionLevel: Math.round((100 - options.quality) / 100 * 9),
  1111→        palette: options.quality < 80,

  1113→      break;
  1114→    case "webp":
  1115→      pipeline = pipeline.webp({ quality: options.quality });
  1116→      break;
  1117→  }
  1118→
  1119→  const buffer = await pipeline.toBuffer();
