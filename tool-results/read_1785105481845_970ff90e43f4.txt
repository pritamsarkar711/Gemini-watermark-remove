   102→   102→  const imgRef = useRef<HTMLImageElement | null>(null)
   103→   103→  const [renderedSize, setRenderedSize] = useState<{ w: number; h: number } | null>(null)
   104→   104→
   105→   105→  useEffect(() => {
   106→   106→    const img = imgRef.current
   107→   107→    if (!img) return
   108→   108→    const measure = () => {
   109→   109→      const r = img.getBoundingClientRect()

   119→   119→      ro.disconnect()

   123→   123→
   124→   124→  // ─── Crop overlay drag state ───────────────────────────────────────────────
   125→   125→  // While dragging, `dragRect` holds the live in-progress rect (image coords);
   126→   126→  // the overlay shows `dragRect ?? cropRect`. On pointer up, dragRect is
   127→   127→  // committed to the store via setCropRect and cleared.
   128→   128→  const [dragRect, setDragRect] = useState<CropRect | null>(null)
   129→   129→  const [isCropDragging, setIsCropDragging] = useState(false)
   130→   130→  const dragStateRef = useRef<DragState | null>(null)
   131→   131→
   132→   132→  /** Convert screen pointer coords to image-space pixel coords. */
   133→   133→  const screenToImage = useCallback(
   134→   134→    (clientX: number, clientY: number): { x: number; y: number } | null => {
   135→   135→      const img = imgRef.current
   136→   136→      if (!img || !originalImage) return null
   137→   137→      const r = img.getBoundingClientRect()
   138→   138→      if (r.width === 0 || r.height === 0) return null
   139→   139→      const x = ((clientX - r.left) / r.width) * originalImage.width
   140→   140→      const y = ((clientY - r.top) / r.height) * originalImage.height
   141→   141→      return { x, y }
   142→   142→    },
   143→   143→    [originalImage]
   144→   144→  )
   145→   145→
   146→   146→  /** Compute the next crop rect given the current pointer position. */
   147→   147→  const computeNextRect = useCallback(
   148→   148→    (pointerImgX: number, pointerImgY: number): CropRect | null => {
   149→   149→      const ds = dragStateRef.current
   150→   150→      if (!ds || !originalImage) return null
   151→   151→      const imgW = originalImage.width
   152→   152→      const imgH = originalImage.height
   153→   153→      const start = ds.startRect
   154→   154→      // Delta in image-space pixels since drag start
   155→   155→      const dx = pointerImgX - ds.startPointerX
   156→   156→      const dy = pointerImgY - ds.startPointerY
   157→   157→
   158→   158→      let { x, y, width, height } = start
   159→   159→
   160→   160→      if (ds.mode === 'move') {
   161→   161→        x = Math.max(0, Math.min(imgW - start.width, start.x + dx))
   162→   162→        y = Math.max(0, Math.min(imgH - start.height, start.y + dy))
   163→   163→      } else {
   164→   164→        // Resize: adjust edges independently, then enforce min size & bounds
   165→   165→        let left = start.x
   166→   166→        let top = start.y
   167→   167→        let right = start.x + start.width
   168→   168→        let bottom = start.y + start.height
   169→   169→
   170→   170→        if (ds.mode.includes('w')) left = start.x + dx
   171→   171→        if (ds.mode.includes('e')) right = start.x + start.width + dx
   172→   172→        if (ds.mode.includes('n')) top = start.y + dy
   173→   173→        if (ds.mode.includes('s')) bottom = start.y + start.height + dy
   174→   174→
   175→   175→        // Clamp to image bounds
   176→   176→        left = Math.max(0, Math.min(left, imgW - MIN_CROP_SIZE))
   177→   177→        right = Math.max(MIN_CROP_SIZE, Math.min(right, imgW))
   178→   178→        top = Math.max(0, Math.min(top, imgH - MIN_CROP_SIZE))
   179→   179→        bottom = Math.max(MIN_CROP_SIZE, Math.min(bottom, imgH))
   180→   180→
   181→   181→        // Enforce min size (swap if user dragged past the opposite edge)
   182→   182→        if (right - left < MIN_CROP_SIZE) {
   183→   183→          // Pin to whichever edge is being dragged
   184→   184→          if (ds.mode.includes('w')) left = right - MIN_CROP_SIZE
   185→   185→          else right = left + MIN_CROP_SIZE
   186→   186→        }
   187→   187→        if (bottom - top < MIN_CROP_SIZE) {
   188→   188→          if (ds.mode.includes('n')) top = bottom - MIN_CROP_SIZE
   189→   189→          else bottom = top + MIN_CROP_SIZE
   190→   190→        }
   191→   191→
   192→   192→        x = left
   193→   193→        y = top
   194→   194→        width = right - left
   195→   195→        height = bottom - top
   196→   196→      }
   197→   197→
   198→   198→      return { x: Math.round(x), y: Math.round(y), width: Math.round(width), height: Math.round(height) }
   199→   199→    },
   200→   200→    [originalImage]
   201→   201→  )
   202→   202→
   203→   203→  // Unified pointer move/up handlers (work for both mouse and touch).
   204→   204→  const handlePointerMove = useCallback(
   205→   205→    (clientX: number, clientY: number) => {
   206→   206→      const next = computeNextRect(clientX, clientY)
   207→   207→      if (next) setDragRect(next)
   208→   208→    },
   209→   209→    [computeNextRect]
   210→   210→  )
   211→   211→
   212→   212→  const handlePointerUp = useCallback(() => {
   213→   213→    // Commit the final rect (if any) to the store, then clear local drag state
   214→   214→    setDragRect((dr) => {
   215→   215→      if (dr) setCropRect(dr)
   216→   216→      return null
   217→   217→    })
   218→   218→    dragStateRef.current = null
   219→   219→    setIsCropDragging(false)
   220→   220→    if (typeof window !== 'undefined') {
   221→   221→      document.body.style.userSelect = ''
   222→   222→    }
   223→   223→  }, [setCropRect])
   224→   224→
   225→   225→  // Attach global listeners during an active drag (so we keep tracking even
   226→   226→  // when the pointer leaves the overlay element).
   227→   227→  useEffect(() => {
   228→   228→    if (!isCropDragging) return
   229→   229→
   230→   230→    const onMouseMove = (e: MouseEvent) => {
   231→   231→      e.preventDefault()
   232→   232→      handlePointerMove(e.clientX, e.clientY)
   233→   233→    }
   234→   234→    const onMouseUp = () => handlePointerUp()
   235→   235→    const onTouchMove = (e: TouchEvent) => {
   236→   236→      if (e.touches.length === 0) return
   237→   237→      e.preventDefault()
   238→   238→      const t = e.touches[0]
   239→   239→      handlePointerMove(t.clientX, t.clientY)
   240→   240→    }
   241→   241→    const onTouchEnd = () => handlePointerUp()
   242→   242→
   243→   243→    window.addEventListener('mousemove', onMouseMove)
   244→   244→    window.addEventListener('mouseup', onMouseUp)
   245→   245→    window.addEventListener('touchmove', onTouchMove, { passive: false })
   246→   246→    window.addEventListener('touchend', onTouchEnd)
   247→   247→    window.addEventListener('touchcancel', onTouchEnd)
   248→   248→    document.body.style.userSelect = 'none'
   249→   249→
   250→   250→    return () => {
   251→   251→      window.removeEventListener('mousemove', onMouseMove)
   252→   252→      window.removeEventListener('mouseup', onMouseUp)
   253→   253→      window.removeEventListener('touchmove', onTouchMove)
   254→   254→      window.removeEventListener('touchend', onTouchEnd)
   255→   255→      window.removeEventListener('touchcancel', onTouchEnd)
   256→   256→    }
   257→   257→  }, [isCropDragging, handlePointerMove, handlePointerUp])
   258→   258→
   259→   259→  /** Begin a drag (move or resize). */
   260→   260→  const beginDrag = useCallback(
   261→   261→    (mode: DragMode, clientX: number, clientY: number) => {
   262→   262→      if (!originalImage) return
   263→   263→      const imgPos = screenToImage(clientX, clientY)
   264→   264→      if (!imgPos) return
   265→   265→      const baseRect = dragRect ?? cropRect
   266→   266→      dragStateRef.current = {
   267→   267→        mode,
   268→   268→        startPointerX: imgPos.x,
   269→   269→        startPointerY: imgPos.y,
   270→   270→        startRect: { ...baseRect },
   271→   271→      }
   272→   272→      // Seed the local drag rect so the overlay snaps to it immediately and so
   273→   273→      // subsequent pointermove events always have a non-null value to update.
   274→   274→      setDragRect({ ...baseRect })
   275→   275→      setIsCropDragging(true)
   276→   276→    },
   277→   277→    [originalImage, screenToImage, dragRect, cropRect]
   278→   278→  )
   279→   279→
   280→   280→  // Mouse + touch handlers for the rect body and each handle.
   281→   281→  const onRectPointerDown = useCallback(
   282→   282→    (e: React.MouseEvent | React.TouchEvent) => {
   283→   283→      if (zoom > 1) return
   284→   284→      e.stopPropagation()
   285→   285→      const pt = 'touches' in e ? e.touches[0] : (e as React.MouseEvent)
   286→   286→      if (!pt) return
   287→   287→      beginDrag('move', pt.clientX, pt.clientY)
   288→   288→    },
   289→   289→    [beginDrag, zoom]
   290→   290→  )
   291→   291→
   292→   292→  const makeHandlePointerDown = useCallback(
   293→   293→    (mode: DragMode) => (e: React.MouseEvent | React.TouchEvent) => {
   294→   294→      if (zoom > 1) return
   295→   295→      e.stopPropagation()
   296→   296→      e.preventDefault()
   297→   297→      const pt = 'touches' in e ? e.touches[0] : (e as React.MouseEvent)
   298→   298→      if (!pt) return
   299→   299→      beginDrag(mode, pt.clientX, pt.clientY)
   300→   300→    },
   301→   301→    [beginDrag, zoom]
   302→   302→  )
   303→   303→
   304→   304→  // ─── Existing zoom / pan / drag-drop handlers ──────────────────────────────
   305→   305→  const handleZoomIn = useCallback(() => {
   306→   306→    setZoom((z) => Math.min(z + 0.5, 5))
   307→   307→  }, [])
   308→   308→
   309→   309→  const handleZoomOut = useCallback(() => {
   310→   310→    setZoom((z) => {
   311→   311→      const newZoom = Math.max(z - 0.5, 1)
   312→   312→      if (newZoom === 1) setOffset({ x: 0, y: 0 })
   313→   313→      return newZoom
   314→   314→    })
   315→   315→  }, [])
   316→   316→
   317→   317→  const handleResetZoom = useCallback(() => {
   318→   318→    setZoom(1)
   319→   319→    setOffset({ x: 0, y: 0 })
   320→   320→  }, [])
   321→   321→
   322→   322→  const handleWheel = useCallback((e: React.WheelEvent) => {
   323→   323→    e.preventDefault()
   324→   324→    setZoom((z) => {
   325→   325→      const newZoom = Math.max(1, Math.min(5, z - e.deltaY * 0.002))
   326→   326→      if (newZoom === 1) setOffset({ x: 0, y: 0 })
   327→   327→      return newZoom
   328→   328→    })
   329→   329→  }, [])
   330→   330→
   331→   331→  const handleMouseDown = useCallback(
   332→   332→    (e: React.MouseEvent) => {
   333→   333→      if (zoom <= 1) return
   334→   334→      setIsDragging(true)
   335→   335→      dragStart.current = { x: e.clientX, y: e.clientY }
   336→   336→      offsetStart.current = { ...offset }
   337→   337→    },
   338→   338→    [zoom, offset]
   339→   339→  )
   340→   340→
   341→   341→  const handleMouseMove = useCallback(
   342→   342→    (e: React.MouseEvent) => {
   343→   343→      if (!isDragging) return
   344→   344→      const dx = e.clientX - dragStart.current.x
   345→   345→      const dy = e.clientY - dragStart.current.y
   346→   346→      setOffset({
   347→   347→        x: offsetStart.current.x + dx,
   348→   348→        y: offsetStart.current.y + dy,
   349→   349→      })
   350→   350→    },
   351→   351→    [isDragging]
   352→   352→  )
   353→   353→
   354→   354→  const handleMouseUp = useCallback(() => {
   355→   355→    setIsDragging(false)
   356→   356→  }, [])
   357→   357→
   358→   358→  // ─── Drag-and-drop re-upload ────────────────────────────────────────────────
   359→   359→  const processFile = useCallback(
   360→   360→    async (file: File) => {
   361→   361→      if (!ACCEPTED_TYPES.includes(file.type)) return
   362→   362→      if (file.size > MAX_SIZE) return
   363→   363→
   364→   364→      const dataUrl = await new Promise<string>((resolve) => {
   365→   365→        const reader = new FileReader()
   366→   366→        reader.onload = () => resolve(reader.result as string)
   367→   367→        reader.readAsDataURL(file)
   368→   368→      })
   369→   369→
   370→   370→      const img = new Image()
   371→   371→      img.onload = () => {
   372→   372→        const imageInfo: ImageInfo = {
   373→   373→          file,
   374→   374→          name: file.name,
   375→   375→          originalName: file.name,
   376→   376→          width: img.naturalWidth,
   377→   377→          height: img.naturalHeight,
   378→   378→          size: file.size,
   379→   379→          type: file.type,
   380→   380→          dataUrl,
   381→   381→        }
   382→   382→        setOriginalImage(imageInfo)
   383→   383→        // Reset zoom/pan when a new image is loaded
   384→   384→        setZoom(1)
   385→   385→        setOffset({ x: 0, y: 0 })
   386→   386→      }
   387→   387→      img.src = dataUrl
   388→   388→    },
   389→   389→    [setOriginalImage]
   390→   390→  )
   391→   391→
   392→   392→  const handleDragOver = useCallback((e: React.DragEvent) => {
   393→   393→    e.preventDefault()
   394→   394→    e.stopPropagation()
   395→   395→    setIsDragOver(true)
   396→   396→  }, [])
   397→   397→
   398→   398→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   399→   399→    e.preventDefault()
   400→   400→    e.stopPropagation()
   401→   401→    // Only clear if leaving the container (not entering a child)
   402→   402→    if (e.currentTarget === e.target) {
   403→   403→      setIsDragOver(false)
   404→   404→    }
   405→   405→  }, [])
   406→   406→
   407→   407→  const handleDrop = useCallback(
   408→   408→    (e: React.DragEvent) => {
   409→   409→      e.preventDefault()
   410→   410→      e.stopPropagation()
   411→   411→      setIsDragOver(false)
   412→   412→      const file = e.dataTransfer.files?.[0]
   413→   413→      if (file) void processFile(file)
   414→   414→    },
   415→   415→    [processFile]
   416→   416→  )
   417→   417→
   418→   418→  if (!originalImage) {
   419→   419→    return (
   420→   420→      <motion.div
   421→   421→        initial={{ opacity: 0, scale: 0.98 }}
   422→   422→        animate={{ opacity: 1, scale: 1 }}
   423→   423→        transition={{ duration: 0.3 }}
   424→   424→        className="flex w-full flex-col gap-2"
   425→   425→      >
   426→   426→        <div
   427→   427→          className="dot-grid-bg flex items-center justify-center rounded-xl border border-dashed bg-muted/20"
   428→   428→          style={{ minHeight: '240px', maxHeight: '55vh' }}
   429→   429→        >
   430→   430→          <div className="flex flex-col items-center gap-2 text-muted-foreground/50">
   431→   431→            <ImageIcon className="size-8" />
   432→   432→            <span className="text-xs font-medium">No image loaded</span>
   433→   433→            <span className="text-[10px] text-muted-foreground/40">Drag and drop or use the upload area</span>
   434→   434→          </div>
   435→   435→        </div>
   436→   436→      </motion.div>
   437→   437→    )
   438→   438→  }
   439→   439→
   440→   440→  // Compute the displayed crop rect (dragRect takes priority while dragging)
   441→   441→  const displayRect = dragRect ?? cropRect
   442→   442→  const hasValidRect = displayRect.width > 0 && displayRect.height > 0
   443→   443→
   444→   444→  // Convert image-space crop rect to screen-space overlay coordinates (as %).
   445→   445→  const overlayPct =
   446→   446→    renderedSize && originalImage && hasValidRect
   447→   447→      ? {
   448→   448→          left: (displayRect.x / originalImage.width) * 100,
   449→   449→          top: (displayRect.y / originalImage.height) * 100,
   450→   450→          width: (displayRect.width / originalImage.width) * 100,
   451→   451→          height: (displayRect.height / originalImage.height) * 100,
   452→   452→        }
   453→   453→      : null
   454→   454→
   455→   455→  const showOverlay =
   456→   456→    isCropOverlayActive && originalImage && overlayPct && zoom <= 1
   457→   457→
   458→   458→  return (
   459→   459→    <motion.div
   460→   460→      initial={{ opacity: 0, scale: 0.98 }}
   461→   461→      animate={{ opacity: 1, scale: 1 }}
   462→   462→      transition={{ duration: 0.3 }}
   463→   463→      className="flex w-full flex-col gap-2"
   464→   464→    >
   465→   465→      {/* Image info bar */}
   466→   466→      <div className="gradient-border-left flex items-center gap-2 rounded-lg border bg-card/80 px-3 py-1.5 shadow-sm transition-all duration-200 hover:bg-card hover:shadow-md">
